WebGL 최적화 설정 ```javascript // webgl-optimization.js function optimizeWebGL() { const canvas = document.createElement('canvas'); const gl = canvas.getContext('webgl2', { alpha: false, // 투명도 비활성화 antialias: false, // 안티앨리어싱 비활성화 depth: false, // 깊이 버퍼 비활성화 failIfMajorPerformanceCaveat: false, powerPreference: "high-performance", premultipliedAlpha: false, preserveDrawingBuffer: false, stencil: false }); // GPU 정보 확인 const debugInfo = gl.getExtension('WEBGL_debug_renderer_info'); if (debugInfo) { console.log('GPU:', gl.getParameter(debugInfo.UNMASKED_RENDERER_WEBGL)); console.log('Driver:', gl.getParameter(debugInfo.UNMASKED_VENDOR_WEBGL)); } return gl; } ``` ### 3. CPU 멀티스레딩 최적화 ```javascript // worker-pool.js class WorkerPool { constructor(workerScript, poolSize = navigator.hardwareConcurrency) { this.workers = []; this.taskQueue = []; this.availableWorkers = []; // 워커 풀 생성 for (let i = 0; i < poolSize; i++) { const worker = new Worker(workerScript); worker._id = i; worker._busy = false; worker.onmessage = (e) => { this.handleWorkerMessage(worker, e); }; this.workers.push(worker); this.availableWorkers.push(worker); } } execute(data) { return new Promise((resolve, reject) => { const task = { data, resolve, reject }; if (this.availableWorkers.length > 0) { this.assignTask(task); } else { this.taskQueue.push(task); } }); } assignTask(task) { const worker = this.availableWorkers.pop(); worker._busy = true; worker._currentTask = task; worker.postMessage(task.data); } handleWorkerMessage(worker, event) { const task = worker._currentTask; worker._busy = false; worker._currentTask = null; if (event.data.error) { task.reject(new Error(event.data.error)); } else { task.resolve(event.data.result); } this.availableWorkers.push(worker); if (this.taskQueue.length > 0) { this.assignTask(this.taskQueue.shift()); } } } ``` ### 4. 메모리 최적화 ```javascript // memory-optimizer.js class MemoryOptimizer { constructor() { this.objectPool = new Map(); this.gcThreshold = 100 * 1024 * 1024; // 100MB this.startMemoryMonitoring(); } // 객체 풀링 getObject(type, ...args) { if (!this.objectPool.has(type)) { this.objectPool.set(type, []); } const pool = this.objectPool.get(type); if (pool.length > 0) { return pool.pop(); } return this.createObject(type, ...args); } releaseObject(type, obj) { // 객체 재사용을 위한 초기화 this.resetObject(obj); const pool = this.objectPool.get(type) || []; if (pool.length < 100) { // 풀 크기 제한 pool.push(obj); } } startMemoryMonitoring() { setInterval(() => { if (performance.memory && performance.memory.usedJSHeapSize > this.gcThreshold) { this.forceGC(); } }, 5000); } forceGC() { if (window.gc) { window.gc(); } // 객체 풀 정리 for (const [type, pool] of this.objectPool) { if (pool.length > 50) { pool.splice(50); } } } } ``` --- ## 🔧 트러블슈팅 ### 일반적인 문제 해결 #### 1.
