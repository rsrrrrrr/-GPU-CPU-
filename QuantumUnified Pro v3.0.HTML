# GPU 지원 확인 (WebGL 2.0 필수) navigator.gpu ? console.log("WebGPU 지원됨") : console.log("WebGL만 지원됨") # CPU 코어 수 확인 navigator.hardwareConcurrency // 논리적 프로세서 수

2. Web Workers 멀티스레딩 활성화

// 자동으로 CPU 코어 수에 맞춰 워커 스레드 생성 const workerCount = Math.min(navigator.hardwareConcurrency || 4, 8); console.log(`활성 워커 스레드: ${workerCount}개`);

3. GPU 가속 최적화 설정

// WebGL 2.0 컨텍스트 최적화 const canvas = document.createElement('canvas'); const gl = canvas.getContext('webgl2', { powerPreference: "high-performance", antialias: false, alpha: false });

4. 메모리 & 전력 관리

// 가비지 컬렉션 최적화 if (window.gc) window.gc(); // 개발 모드에서만 // 배터리 API 활용 (지원 시) if ('getBattery' in navigator) { navigator.getBattery().then(battery => { console.log(`배터리 레벨: ${battery.level * 100}%`); }); }
